// 11. Crear un array de estudiantes con materias y calcular promedios

const estudiantesNotas = [
    {
        nombre: "Ana",
        semestre: 3,
        materias: [
            {nombre: "Matematicas", nota: 4.0},
            {nombre: "Programacion", nota: 4.5},
            {nombre: "Bases de datos", nota: 3.8}
        ]
    },
    {
        nombre: "Carlos",
        semestre: 5,
        materias: [
            {nombre: "Matematicas", nota: 3.0},
            {nombre: "Programacion", nota: 3.5},
            {nombre: "Bases de datos", nota: 4.0}
        ]
    }
];

let sumaPromedios = 0;

for (let i = 0; i < estudiantesNotas.length; i++) {

    let sumaNotas = 0;

    for (let j = 0; j < estudiantesNotas[i].materias.length; j++) {
        sumaNotas += estudiantesNotas[i].materias[j].nota;
    }

    let promedioEstudiante = sumaNotas / estudiantesNotas[i].materias.length;

    console.log("Promedio de", estudiantesNotas[i].nombre, ":", promedioEstudiante);

    sumaPromedios += promedioEstudiante;
}

let promedioGeneral = sumaPromedios / estudiantesNotas.length;

console.log("Promedio general:", promedioGeneral);
